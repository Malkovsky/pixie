cmake_minimum_required(VERSION 3.18)
project(pixie_microtree_simd_notes LANGUAGES CXX)

enable_testing()
get_filename_component(PIXIE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
option(MICRO_BUILD_BENCHMARKS "Build the archived timing probes" OFF)

add_executable(probe probe.cpp)
target_compile_features(probe PRIVATE cxx_std_20)
target_include_directories(probe PRIVATE "${PIXIE_ROOT}" "${PIXIE_ROOT}/include")
add_test(NAME exact_traces COMMAND probe)

if(MICRO_BUILD_BENCHMARKS)
  find_package(benchmark CONFIG REQUIRED)
  target_link_libraries(probe PRIVATE benchmark::benchmark)
  add_executable(kernels kernels.cpp)
  target_compile_features(kernels PRIVATE cxx_std_20)
  target_include_directories(kernels PRIVATE "${PIXIE_ROOT}" "${PIXIE_ROOT}/include")
  target_link_libraries(kernels PRIVATE benchmark::benchmark)
else()
  target_compile_definitions(probe PRIVATE MICRO_CHECK_ONLY)
endif()
